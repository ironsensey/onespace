// LoadCACertificate loads and validates a CA certificate from the given path.
// Returns a certificate pool with the loaded certificate and the SHA-256 fingerprint.
func LoadCACertificate(certPath string) (*x509.CertPool, string, error) {
	if certPath == "" {
		return nil, "", fmt.Errorf("certificate path is empty")
	}

	caCertPEM, err := os.ReadFile(certPath)
	if err != nil {
		return nil, "", fmt.Errorf("failed to read CA certificate from %s: %w", certPath, err)
	}
	if len(caCertPEM) == 0 {
		return nil, "", fmt.Errorf("CA certificate file is empty: %s", certPath)
	}

	caCertPool := x509.NewCertPool()
	if !caCertPool.AppendCertsFromPEM(caCertPEM) {
		return nil, "", fmt.Errorf("failed to parse CA certificate from %s: malformed PEM or invalid X.509 certificate", certPath)
	}

	certs := caCertPool.Subjects()
	if len(certs) == 0 {
		return nil, "", fmt.Errorf("no certificates accepted into trust pool from %s", certPath)
	}

	block, _ := pem.Decode(caCertPEM)
	if block == nil || block.Type != "CERTIFICATE" {
		return nil, "", fmt.Errorf("no valid CERTIFICATE block found in %s", certPath)
	}

	parsedCert, err := x509.ParseCertificate(block.Bytes)
	if err != nil {
		return nil, "", fmt.Errorf("failed to parse certificate for fingerprint: %w", err)
	}

	fp := sha256.Sum256(parsedCert.Raw)
	fingerprint := hex.EncodeToString(fp[:])
	return caCertPool, fingerprint, nil
}

func run() error {
	flag.Parse()

	var kp wireguard.KeyPair
	var err error

	// Validate client configuration before creating resources
	if *mode == "client" && *caCertFile == "" {
		return fmt.Errorf("-ca-cert flag is required for client mode")
	}

	if *keyFile != "" {
		kp, err = loadOrGenerateKeyPair(*keyFile)
		if err != nil {
			return fmt.Errorf("key file: %w", err)
		}
		fmt.Fprintf(os.Stderr, "Loaded persistent key pair\n")
	} else {
		kp, err = wireguard.GenerateKeyPair()
		if err != nil {
			return fmt.Errorf("generate key: %w", err)
		}
		os.Stderr.WriteString("Generated WireGuard key pair\n")
	}
	fmt.Printf("PUBLIC_KEY=%s\n", kp.PublicKey)
	fmt.Printf("PRIVATE_KEY=%s\n", kp.PrivateKey)
	fmt.Printf("MODE=%s\n", *mode)

	// Create tunnel
	tun, err := wireguard.NewTunnel(wireguard.TunnelConfig{
		Name:     "wg-test",
		MTU:      1420,
		LogLevel: 0,
	})
	if err != nil {
		return fmt.Errorf("create tunnel: %w", err)
	}
	defer tun.Close()

	// Configure interface
	cfg := wireguard.Config{
		PrivateKey: kp.PrivateKey,
		Address:    *overlayIP,
		ListenPort: *listenPort,
		MTU:        1420,
	}
	if err := tun.Configure(cfg); err != nil {
		return fmt.Errorf("configure tunnel: %w", err)
	}
	fmt.Println("TUNNEL_UP")

	// Configure interface IP address on Linux
	if err := configureInterfaceIP(*overlayIP); err != nil {
		fmt.Fprintf(os.Stderr, "configure interface IP: %v\n", err)
	}

	// Add peer if provided
	if *peerKey != "" {
		peer := wireguard.Peer{
			PublicKey:           *peerKey,
			Endpoint:            fmt.Sprintf("%s:%d", *peerIP, *peerPort),
			AllowedIPs:          []string{*peerOverlayIP + "/32"},
			PersistentKeepalive: 1,
		}
		cfg.Peers = []wireguard.Peer{peer}
		if err := tun.Configure(cfg); err != nil {
			return fmt.Errorf("configure peer: %w", err)
		}
		fmt.Println("PEER_ADDED")
	}

	if *mode == "server" {
		return runServer(tun)
	} else if *mode == "client" {
		return runClient(tun, *enrollTimeout)
	}
	return fmt.Errorf("specify -mode server or -mode client")
}

func configureInterfaceIP(overlayIP string) error {
	if runtime.GOOS != "linux" {
		return nil // Windows handles IP via netsh in client code
	}
	// Parse IP and prefix
	ipParts := overlayIP
	if idx := indexOf(ipParts, '/'); idx >= 0 {
		ipParts = ipParts[:idx]
	}
	// Bring interface up
	if err := exec.Command("ip", "link", "set", "wg-test", "up").Run(); err != nil {
		return fmt.Errorf("ip link set up: %w", err)
	}
	// Add IP address
	if err := exec.Command("ip", "addr", "add", overlayIP, "dev", "wg-test").Run(); err != nil {
		return fmt.Errorf("ip addr add: %w", err)
	}
	return nil
}

func loadOrGenerateKeyPair(path string) (wireguard.KeyPair, error) {
	if data, err := os.ReadFile(path); err == nil {
		// Load existing key (hex private key)
		privHex := string(data)
		return wireguard.LoadKeyPair(privHex)
	}

	// Generate new key
	kp, err := wireguard.GenerateKeyPair()
	if err != nil {
		return wireguard.KeyPair{}, err
	}
	// Write private key (hex) to file
	if err := os.WriteFile(path, []byte(kp.PrivateKey), 0600); err != nil {
		return wireguard.KeyPair{}, fmt.Errorf("write key file: %w", err)
	}
	return kp, nil
}

func runServer(tun *wireguard.Tunnel) error {
	// Start HTTPS server on overlay IP
	tlsCert, serverCert, err := generateTestCert()
	if err != nil {
		return fmt.Errorf("generate cert: %w", err)
	}

	overlayAddr := *overlayIP
	if idx := indexOf(overlayAddr, '/'); idx >= 0 {
		overlayAddr = overlayAddr[:idx]
	}

	server := &http.Server{
		Addr:      overlayAddr + ":8443",
		TLSConfig: &tls.Config{Certificates: []tls.Certificate{*tlsCert}},
		Handler:   http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			w.Write([]byte("ok"))
		}),
	}
	go server.ListenAndServeTLS("", "")
	fmt.Printf("HTTPS_SERVER_STARTED on %s:8443\n", overlayAddr)

	// Store server cert for client trust (write to file)
	if serverCert != nil {
		certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: serverCert.Raw})
		os.WriteFile("/tmp/server-cert.pem", certPEM, 0644)
	}

	// Handle revoke
	if *revoke {
		if err := tun.Device.IpcSet("public_key=" + *revokeKey + "\nremove=true\n"); err != nil {
			fmt.Fprintf(os.Stderr, "revoke: %v\n", err)
		} else {
			fmt.Println("PEER_REVOKED")
		}
	}

	// Wait for signal
	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)
	<-sigCh
	fmt.Println("SHUTTING_DOWN")
	return nil
}

func runClient(tun *wireguard.Tunnel, enrollTimeout time.Duration) error {
	overlayAddr := *overlayIP
	if idx := indexOf(overlayAddr, '/'); idx >= 0 {
		overlayAddr = overlayAddr[:idx]
	}
	peerOverlayAddr := *peerOverlayIP

	// Wait for tunnel to establish and peer to be configured
	fmt.Printf("ENROLL_WAIT_START timeout=%v\n", enrollTimeout)
	time.Sleep(enrollTimeout)
	fmt.Println("ENROLL_WAIT_DONE")

	// Load server cert for validation
	certPath := *caCertFile // Already validated in run()

	caCertPool, fingerprint, err := LoadCACertificate(certPath)
	if err != nil {
		return fmt.Errorf("load CA certificate: %w", err)
	}
	fmt.Fprintf(os.Stderr, "Loaded and validated CA certificate from %s (SHA-256: %s)\n", certPath, fingerprint)

	// Make HTTPS request through tunnel with certificate validation
	client := &http.Client{
		Timeout: 10 * time.Second,
		Transport: &http.Transport{
			TLSClientConfig: &tls.Config{
				RootCAs:    caCertPool,
				ServerName: "test-receiver",
			},
		},
	}

	req, _ := http.NewRequest("GET", "https://"+peerOverlayAddr+":8443/", nil)
	resp, err := client.Do(req)
	if err != nil {
		fmt.Printf("REQUEST_FAILED: %v\n", err)
	} else {
		body, _ := io.ReadAll(resp.Body)
		resp.Body.Close()
		fmt.Printf("REQUEST_RESULT: status=%d body=%s\n", resp.StatusCode, string(body))
	}

	// Wait a bit then check if we should revoke
	time.Sleep(2 * time.Second)
	fmt.Println("CLIENT_DONE")
	return nil
}

func generateTestCert() (*tls.Certificate, *x509.Certificate, error) {
	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return nil, nil, err
	}
	template := &x509.Certificate{
		SerialNumber: big.NewInt(1),
		Subject:      pkix.Name{CommonName: "test-receiver"},
		NotBefore:    time.Now(),
		NotAfter:     time.Now().Add(time.Hour),
		KeyUsage:     x509.KeyUsageDigitalSignature,
		ExtKeyUsage:  []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
		IPAddresses:  []net.IP{net.IPv4(10, 200, 200, 2)},
		DNSNames:     []string{"test-receiver"},
	}
	certDER, err := x509.CreateCertificate(rand.Reader, template, template, &key.PublicKey, key)
	if err != nil {
		return nil, nil, err
	}
	certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: certDER})
	keyPEM := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: x509.MarshalPKCS1PrivateKey(key)})
	cert, err := tls.X509KeyPair(certPEM, keyPEM)
	if err != nil {
		return nil, nil, err
	}
	// Parse cert for client trust
	parsedCert, _ := x509.ParseCertificate(certDER)
	return &cert, parsedCert, nil
}

func indexOf(s string, c byte) int {
	for i := 0; i < len(s); i++ {
		if s[i] == c {
			return i
		}
	}
	return -1
}

// TestLoadCACertificate tests the certificate loading functionality with various scenarios
func TestLoadCACertificate(t *testing.T) {
	tests := []struct {
		name          string
		certPEM       []byte
		expectError   bool
		errorContains string
	}{
		{
			name: "valid certificate",
			certPEM: func() []byte {
				key, _ := rsa.GenerateKey(rand.Reader, 2048)
				template := &x509.Certificate{
					SerialNumber: big.NewInt(1),
					Subject:      pkix.Name{CommonName: "test-ca"},
					NotBefore:    time.Now(),
					NotAfter:     time.Now().Add(time.Hour),
					KeyUsage:     x509.KeyUsageCertSign | x509.KeyUsageDigitalSignature,
					ExtKeyUsage:  []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
					IsCA:         true,
					BasicConstraintsValid: true,
				}
				certDER, _ := x509.CreateCertificate(rand.Reader, template, template, &key.PublicKey, key)
				return pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: certDER})
			}(),
			expectError: false,
		},
		{
			name: "missing file",
			certPEM:       nil,
			expectError:   true,
			errorContains: "cannot find the file specified",
		},
		{
			name: "empty file",
			certPEM:       []byte(""),
			expectError:   true,
			errorContains: "CA certificate file is empty",
		},
		{
			name: "malformed PEM",
			certPEM:       []byte("not a valid certificate"),
			expectError:   true,
			errorContains: "malformed PEM or invalid X.509 certificate",
		},
		{
			name: "valid PEM but not a certificate",
			certPEM: func() []byte {
				key, _ := rsa.GenerateKey(rand.Reader, 2048)
				keyPEM := pem.EncodeToMemory(&pem.Block{Type: "RSA PRIVATE KEY", Bytes: x509.MarshalPKCS1PrivateKey(key)})
				return keyPEM
			}(),
			expectError:   true,
			errorContains: "malformed PEM or invalid X.509 certificate",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			// Create a temporary file for the certificate
			tmpDir := t.TempDir()
			certPath := filepath.Join(tmpDir, "ca.pem")

			if tt.certPEM != nil {
				err := os.WriteFile(certPath, tt.certPEM, 0644)
				if err != nil {
					t.Fatalf("failed to write test certificate: %v", err)
				}
			}

			pool, fingerprint, err := LoadCACertificate(certPath)

			if tt.expectError {
				if err == nil {
					t.Errorf("expected error containing %q, got nil", tt.errorContains)
				} else if !contains(err.Error(), tt.errorContains) {
					t.Errorf("expected error containing %q, got %q", tt.errorContains, err.Error())
				}
				if pool != nil {
					t.Errorf("expected nil pool on error, got %v", pool)
				}
				if fingerprint != "" {
					t.Errorf("expected empty fingerprint on error, got %s", fingerprint)
				}
			} else {
				if err != nil {
					t.Errorf("unexpected error: %v", err)
				}
				if pool == nil {
					t.Errorf("expected non-nil pool, got nil")
				}
				if fingerprint == "" {
					t.Errorf("expected non-empty fingerprint")
				}
				// Verify fingerprint format (64 hex characters for SHA-256)
				if len(fingerprint) != 64 {
					t.Errorf("expected 64-character hex fingerprint, got %d: %s", len(fingerprint), fingerprint)
				}
				// Verify fingerprint is valid hex
				if _, err := hex.DecodeString(fingerprint); err != nil {
					t.Errorf("fingerprint is not valid hex: %v", err)
				}
			}
		})
	}
}

// TestLoadCACertificate_FileErrors tests file system related errors
func TestLoadCACertificate_FileErrors(t *testing.T) {
	tmpDir := t.TempDir()

	// Test non-existent file
	nonExistent := filepath.Join(tmpDir, "nonexistent.pem")
	_, _, err := LoadCACertificate(nonExistent)
	if err == nil {
		t.Error("expected error for non-existent file")
	}

	// Test unreadable file (permission denied)
	unreadable := filepath.Join(tmpDir, "unreadable.pem")
	writeErr := os.WriteFile(unreadable, []byte("test"), 0000)
	if writeErr == nil {
		// Try to read it - should fail
		_, _, err := LoadCACertificate(unreadable)
		if err == nil {
			t.Error("expected error for unreadable file")
		}
		// Restore permissions for cleanup
		os.Chmod(unreadable, 0644)
	}
}

// TestLoadCACertificate_FingerprintConsistency tests that the same certificate produces the same fingerprint
func TestLoadCACertificate_FingerprintConsistency(t *testing.T) {
	key, _ := rsa.GenerateKey(rand.Reader, 2048)
	template := &x509.Certificate{
		SerialNumber: big.NewInt(1),
		Subject:      pkix.Name{CommonName: "test-ca"},
		NotBefore:    time.Now(),
		NotAfter:     time.Now().Add(time.Hour),
		KeyUsage:     x509.KeyUsageCertSign | x509.KeyUsageDigitalSignature,
		ExtKeyUsage:  []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
		IsCA:         true,
		BasicConstraintsValid: true,
	}
	certDER, _ := x509.CreateCertificate(rand.Reader, template, template, &key.PublicKey, key)
	certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: certDER})

	tmpDir := t.TempDir()
	certPath := filepath.Join(tmpDir, "ca.pem")
	err := os.WriteFile(certPath, certPEM, 0644)
	if err != nil {
		t.Fatalf("failed to write test certificate: %v", err)
	}

	// Load twice and verify fingerprint consistency
	_, fp1, err := LoadCACertificate(certPath)
	if err != nil {
		t.Fatalf("first load failed: %v", err)
	}
	_, fp2, err := LoadCACertificate(certPath)
	if err != nil {
		t.Fatalf("second load failed: %v", err)
	}
	if fp1 != fp2 {
		t.Errorf("fingerprint mismatch: %s vs %s", fp1, fp2)
	}

	// Verify fingerprint matches direct SHA-256 of DER
	parsedCert, _ := x509.ParseCertificate(certDER)
	sum := sha256.Sum256(parsedCert.Raw)
	expectedFP := hex.EncodeToString(sum[:])
	if fp1 != expectedFP {
		t.Errorf("fingerprint mismatch with direct calculation: %s vs %s", fp1, expectedFP)
	}
}

// contains checks if a string contains a substring
func contains(s, substr string) bool {
	return len(s) >= len(substr) && (s == substr || len(s) > 0 && findSubstring(s, substr))
}

func findSubstring(s, substr string) bool {
	for i := 0; i <= len(s)-len(substr); i++ {
		if s[i:i+len(substr)] == substr {
			return true
		}
	}
	return false
}

package wireguard

import (
	"crypto/ecdh"
	"crypto/rand"
	"encoding/base64"
	"encoding/hex"
	"fmt"
)

// KeyPair holds a WireGuard public/private key pair.
// PrivateKey and PublicKey are hex-encoded for wireguard-go IPC compatibility.
// Use Base64PrivateKey/Base64PublicKey for standard WireGuard format.
type KeyPair struct {
	PrivateKey       string // hex-encoded (wireguard-go IPC format)
	PublicKey        string // hex-encoded (wireguard-go IPC format)
	Base64PrivateKey string // base64-encoded (standard WireGuard format)
	Base64PublicKey  string // base64-encoded (standard WireGuard format)
}

// GenerateKeyPair generates a new WireGuard key pair using crypto/ecdh X25519.
// Returns hex-encoded keys for wireguard-go IPC and base64 for standard WireGuard.
func GenerateKeyPair() (KeyPair, error) {
	curve := ecdh.X25519()
	priv, err := curve.GenerateKey(rand.Reader)
	if err != nil {
		return KeyPair{}, fmt.Errorf("generate key: %w", err)
	}
	privBytes := priv.Bytes()
	pubBytes := priv.PublicKey().Bytes()

	return KeyPair{
		PrivateKey:       hex.EncodeToString(privBytes),
		PublicKey:        hex.EncodeToString(pubBytes),
		Base64PrivateKey: base64.StdEncoding.EncodeToString(privBytes),
		Base64PublicKey:  base64.StdEncoding.EncodeToString(pubBytes),
	}, nil
}

// LoadKeyPair loads a WireGuard key pair from a hex-encoded private key.
func LoadKeyPair(privHex string) (KeyPair, error) {
	privBytes, err := hex.DecodeString(privHex)
	if err != nil {
		return KeyPair{}, fmt.Errorf("decode key: %w", err)
	}
	if len(privBytes) != 32 {
		return KeyPair{}, fmt.Errorf("invalid key length: %d", len(privBytes)
	}
	curve := ecdh.X25519()
	priv, err := curve.NewPrivateKey(privBytes)
	if err != nil {
		return KeyPair{}, fmt.Errorf("load private key: %w", err)
	}
	pubBytes := priv.PublicKey().Bytes()
	return KeyPair{
		PrivateKey:       hex.EncodeToString(privBytes),
		PublicKey:        hex.EncodeToString(pubBytes),
		Base64PrivateKey: base64.StdEncoding.EncodeToString(privBytes),
		Base64PublicKey:  base64.StdEncoding.EncodeToString(pubBytes),
	}, nil
}

// Base64Encode encodes bytes to base64.
func Base64Encode(data []byte) string {
	return base64.StdEncoding.EncodeToString(data)
}

package wireguard

import (
	"fmt"
	"time"

	"golang.zx2c4.com/wireguard/conn"
	"golang.zx2c4.com/wireguard/device"
	"golang.zx2c4.com/wireguard/tun"
)

// Tunnel wraps a WireGuard device with its TUN interface.
type Tunnel struct {
	Device  *device.Device
	TUN     tun.Device
	Bind    conn.Bind
	Logger  *device.Logger
	Name    string
}

// TunnelConfig holds configuration for creating a tunnel.
type TunnelConfig struct {
	Name     string
	MTU      int
	LogLevel int
}

// NewTunnel creates a new WireGuard tunnel with a TUN device.
func NewTunnel(cfg TunnelConfig) (*Tunnel, error) {
	if cfg.MTU == 0 {
		cfg.MTU = 1420
	}
	if cfg.Name == "" {
		cfg.Name = "wg0"
	}

	tunDev, err := tun.CreateTUN(cfg.Name, cfg.MTU)
	if err != nil {
		return nil, fmt.Errorf("create TUN: %w", err)
	}

	bind := conn.NewDefaultBind()
	logger := device.NewLogger(cfg.LogLevel, "[wireguard]")

	dev := device.NewDevice(tunDev, bind, logger)

	t := &Tunnel{
		Device: dev,
		TUN:    tunDev,
		Bind:   bind,
		Logger: logger,
		Name:   cfg.Name,
	}

	return t, nil
}

// Configure applies a WireGuard configuration to the tunnel.
func (t *Tunnel) Configure(cfg Config) error {
	uapi := fmt.Sprintf("private_key=%s\n", cfg.PrivateKey)
	if cfg.ListenPort > 0 {
		uapi += fmt.Sprintf("listen_port=%d\n", cfg.ListenPort)
	}

	for _, peer := range cfg.Peers {
		uapi += fmt.Sprintf("public_key=%s\n", peer.PublicKey)
		if peer.Endpoint != "" {
			uapi += fmt.Sprintf("endpoint=%s\n", peer.Endpoint)
		}
		if len(peer.AllowedIPs) > 0 {
			for _, ip := range peer.AllowedIPs {
				uapi += fmt.Sprintf("allowed_ip=%s\n", ip)
			}
		}
		if peer.PersistentKeepalive > 0 {
			uapi += fmt.Sprintf("persistent_keepalive_interval=%d\n", peer.PersistentKeepalive)
		}
	}

	if err := t.Device.IpcSet(uapi); err != nil {
		return fmt.Errorf("configure device: %w", err)
	}

	// Bring interface up
	if err := t.Device.Up(); err != nil {
		return fmt.Errorf("bring up: %w", err)
	}

	return nil
}

// Close shuts down the tunnel.
func (t *Tunnel) Close() {
	if t.Device != nil {
		t.Device.Close()
	}
	if t.Bind != nil {
		t.Bind.Close()
	}
}

// GetStats returns transfer statistics.
func (t *Tunnel) GetStats() (uint64, uint64, error) {
	stats, err := t.Device.IpcGet()
	if err != nil {
		return 0, 0, err
	}
	// Parse stats from IPC output
	_ = stats
	_ = time.Second
	return 0, 0, nil
}

// Package wireguard provides optional integrated WireGuard connectivity for OneSpace.
// It generates and manages WireGuard configurations, handles enrollment,
// peer discovery, and direct/relay selection. It does not implement cryptography
// itself; it uses the system's WireGuard installation (wireguard-tools on Linux,
// wireguard-nt on Windows).
package wireguard

import (
	"fmt"
	"net"
	"strings"
)


// Peer represents a WireGuard peer.
type Peer struct {
	PublicKey           string
	Endpoint            string // host:port
	AllowedIPs          []string
	PersistentKeepalive int
	LastHandshake       string
	TransferRx          uint64
	TransferTx          uint64
}

// Config represents a WireGuard interface configuration.
type Config struct {
	PrivateKey string
	Address    string // e.g., "10.200.200.2/24"
	ListenPort int
	Peers      []Peer
	MTU        int
	DNS        []string
}

// RenderConfig produces a WireGuard configuration file content.
func (c *Config) RenderConfig() string {
	var b strings.Builder
	b.WriteString("[Interface]\n")
	b.WriteString("PrivateKey = " + c.PrivateKey + "\n")
	if c.Address != "" {
		b.WriteString("Address = " + c.Address + "\n")
	}
	if c.ListenPort > 0 {
		b.WriteString("ListenPort = " + fmt.Sprintf("%d", c.ListenPort) + "\n")
	}
	if c.MTU > 0 {
		b.WriteString("MTU = " + fmt.Sprintf("%d", c.MTU) + "\n")
	}
	for _, dns := range c.DNS {
		b.WriteString("DNS = " + dns + "\n")
	}
	b.WriteString("\n")

	for _, peer := range c.Peers {
		b.WriteString("[Peer]\n")
		b.WriteString("PublicKey = " + peer.PublicKey + "\n")
		if peer.Endpoint != "" {
			b.WriteString("Endpoint = " + peer.Endpoint + "\n")
		}
		if len(peer.AllowedIPs) > 0 {
			b.WriteString("AllowedIPs = " + strings.Join(peer.AllowedIPs, ", ") + "\n")
		}
		if peer.PersistentKeepalive > 0 {
			b.WriteString("PersistentKeepalive = " + fmt.Sprintf("%d", peer.PersistentKeepalive) + "\n")
		}
		b.WriteString("\n")
	}

	return b.String()
}

// ConnectionMode represents the current connection mode.
type ConnectionMode int

const (
	ModeLAN ConnectionMode = iota
	ModeWireGuardDirect
	ModeWireGuardRelay
	ModeOffline
	ModeCoordinatorRelay // existing WebRTC/TURN fallback
)

func (m ConnectionMode) String() string {
	switch m {
	case ModeLAN:
		return "lan"
	case ModeWireGuardDirect:
		return "wireguard-direct"
	case ModeWireGuardRelay:
		return "wireguard-relay"
	case ModeOffline:
		return "offline"
	case ModeCoordinatorRelay:
		return "coordinator-relay"
	default:
		return "unknown"
	}
}

// PeerInfo holds discovery information for a peer.
type PeerInfo struct {
	DeviceID      string
	PublicKey     string
	Endpoints     []string // candidate endpoints
	RelayEndpoint string   // relay server endpoint if available
	LastSeen      string
}

// NetworkStatus holds the current network status.
type NetworkStatus struct {
	Mode          ConnectionMode
	PeerCount     int
	DirectPeers   int
	RelayPeers    int
	BytesReceived uint64
	BytesSent     uint64
	LastHandshake string
	MTU           int
}

// AllocateAddress assigns a unique IP address from the overlay subnet.
func AllocateAddress(subnet string, used []string) (string, error) {
	_, ipnet, err := net.ParseCIDR(subnet)
	if err != nil {
		return "", fmt.Errorf("parse subnet: %w", err)
	}

	usedSet := make(map[string]bool)
	for _, ip := range used {
		usedSet[ip] = true
	}

	// Start from .2 (reserve .1 for gateway)
	ip := ipnet.IP.Mask(ipnet.Mask)
	ip[3] = 2

	for ipnet.Contains(ip) {
		ipStr := ip.String()
		if !usedSet[ipStr] {
			return ipStr + "/32", nil
		}
		ip[3]++
		if ip[3] == 0 {
			ip[2]++
		}
	}

	return "", fmt.Errorf("no available addresses in subnet %s", subnet)
}

		Address:    "10.200.200.2/24",

// TestWireGuardEndToEnd creates two real WireGuard peers, performs a handshake,
// sends an authenticated HTTPS request through the tunnel, revokes one peer,
// and verifies traffic stops.
func TestWireGuardEndToEnd(t *testing.T) {
	// Generate keys for both peers
	kp1, err := GenerateKeyPair()
	if err != nil {
		t.Fatalf("generate peer1 key: %v", err)
	}
	kp2, err := GenerateKeyPair()
	if err != nil {
		t.Fatalf("generate peer2 key: %v", err)
	}

	// Create peer 1 tunnel (listens on 51821)
	tun1, err := tun.CreateTUN("wg-test1", 1420)
	if err != nil {
		t.Skipf("cannot create TUN device: %v", err)
	}
	defer tun1.Close()

	bind1 := conn.NewDefaultBind()
	dev1 := device.NewDevice(tun1, bind1, device.NewLogger(device.LogLevelSilent, "[wg1]"))
	defer dev1.Close()

	cfg1 := fmt.Sprintf("private_key=%s\nlisten_port=51821\n", kp1.PrivateKey)
	if err := dev1.IpcSet(cfg1); err != nil {
		t.Fatalf("configure peer1: %v", err)
	}
	if err := dev1.Up(); err != nil {
		t.Fatalf("bring up peer1: %v", err)
	}

	// Create peer 2 tunnel (listens on 51822)
	tun2, err := tun.CreateTUN("wg-test2", 1420)
	if err != nil {
		t.Skipf("cannot create TUN device: %v", err)
	}
	defer tun2.Close()

	bind2 := conn.NewDefaultBind()
	dev2 := device.NewDevice(tun2, bind2, device.NewLogger(device.LogLevelSilent, "[wg2]"))
	defer dev2.Close()

	cfg2 := fmt.Sprintf("private_key=%s\nlisten_port=51822\n", kp2.PrivateKey)
	if err := dev2.IpcSet(cfg2); err != nil {
		t.Fatalf("configure peer2: %v", err)
	}
	if err := dev2.Up(); err != nil {
		t.Fatalf("bring up peer2: %v", err)
	}

	// Configure peer1 -> peer2
	cfg1p := fmt.Sprintf(
		"public_key=%s\nendpoint=127.0.0.1:51822\nallowed_ip=10.200.200.2/32\npersistent_keepalive_interval=1\n",
		kp2.PublicKey,
	)
	if err := dev1.IpcSet(cfg1p); err != nil {
		t.Fatalf("configure peer1 peer: %v", err)
	}

	// Configure peer2 -> peer1
	cfg2p := fmt.Sprintf(
		"public_key=%s\nendpoint=127.0.0.1:51821\nallowed_ip=10.200.200.1/32\npersistent_keepalive_interval=1\n",
		kp1.PublicKey,
	)
	if err := dev2.IpcSet(cfg2p); err != nil {
		t.Fatalf("configure peer2 peer: %v", err)
	}

	// Wait for handshake
	time.Sleep(3 * time.Second)

	// Start HTTPS server on peer 2's overlay IP
	tlsCert, err := generateTestCert()
	if err != nil {
		t.Fatalf("generate test cert: %v", err)
	}
	server := &http.Server{
		Addr:      "10.200.200.2:8443",
		TLSConfig: &tls.Config{Certificates: []tls.Certificate{*tlsCert}},
		Handler:   http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			w.Write([]byte("ok"))
		}),
	}
	go server.ListenAndServeTLS("", "")
	defer server.Close()

	// Wait for server to be ready
	time.Sleep(1 * time.Second)

	// Capture peer2 rx bytes before traffic
	ipc2Before, err := dev2.IpcGet()
	if err != nil {
		t.Fatalf("get peer2 stats: %v", err)
	}
	rxBefore := parseRxBytes(ipc2Before, kp1.PublicKey)
	t.Logf("peer2 rx bytes before traffic: %d", rxBefore)

	// Make HTTPS request from peer 1 through tunnel
	client := &http.Client{
		Timeout: 10 * time.Second,
		Transport: &http.Transport{
			TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
			DialContext: func(ctx context.Context, network, addr string) (net.Conn, error) {
				localAddr := &net.TCPAddr{IP: net.ParseIP("10.200.200.1"), Port: 0}
				d := &net.Dialer{LocalAddr: localAddr}
				return d.DialContext(ctx, network, addr)
			},
		},
	}

	req, err := http.NewRequest("GET", "https://10.200.200.2:8443/", nil)
	if err != nil {
		t.Fatalf("create request: %v", err)
	}

	resp, err := client.Do(req)
	if err != nil {
		t.Fatalf("request through tunnel failed: %v", err)
	}
	body, _ := io.ReadAll(resp.Body)
	resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		t.Errorf("expected 200, got %d", resp.StatusCode)
	}
	if string(body) != "ok" {
		t.Errorf("expected 'ok', got %q", string(body)
	}
		Log("HTTPS request through WireGuard tunnel succeeded")

	// Verify peer2 received bytes increased (traffic went through tunnel)
	ipc2After, err := dev2.IpcGet()
	if err != nil {
		t.Fatalf("get peer2 stats after: %v", err)
	}
	rxAfter := parseRxBytes(ipc2After, kp1.PublicKey)
	t.Logf("peer2 rx bytes after traffic: %d", rxAfter)
	if rxAfter <= rxBefore {
		t.Errorf("peer2 rx bytes did not increase: before=%d after=%d", rxBefore, rxAfter)
	}

	// Revoke peer 1 from peer 2's device using remove=true
	if err := dev2.IpcSet("public_key=" + kp1.PublicKey + "\nremove=true\n"); err != nil {
		t.Fatalf("revoke peer1 from peer2: %v", err)
	}
	t.Log("peer1 revoked from peer2 via remove=true")

	// Wait for revocation to take effect
	time.Sleep(2 * time.Second)

	// Verify peer1 no longer in peer2's peer list
	ipc2AfterRevoke, err := dev2.IpcGet()
	if err != nil {
		t.Fatalf("get peer2 stats after revoke: %v", err)
	}
	if strings.Contains(ipc2AfterRevoke, "public_key="+kp1.PublicKey) {
		t.Error("peer1 still present in peer2 after revocation")
	}
	t.Log("peer1 removed from peer2 peer list")

	// Verify request fails after revocation using a FRESH connection
	resp2, err := client.Do(req)
	if err == nil {
		resp2.Body.Close()
		t.Error("expected request to fail after revocation, but it succeeded")
	} else {
		t.Logf("request correctly failed after revocation: %v", err)
	}

	t.Log("WireGuard end-to-end test completed")
}

// generateTestCert creates a self-signed test certificate.
func generateTestCert() (*tls.Certificate, error) {
	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return nil, err
	}
	template := &x509.Certificate{
		SerialNumber: big.NewInt(1),
		Subject:      pkix.Name{CommonName: "test-receiver"},
		NotBefore:    time.Now(),
		NotAfter:     time.Now().Add(time.Hour),
		KeyUsage:     x509.KeyUsageDigitalSignature,
		ExtKeyUsage:  []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
	}
	certDER, err := x509.CreateCertificate(rand.Reader, template, template, &key.PublicKey, key)
	if err != nil {
		return nil, err
	}
	certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: certDER})
	keyPEM := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: x509.MarshalPKCS1PrivateKey(key)})
	cert, err := tls.X509KeyPair(certPEM, keyPEM)
	if err != nil {
		return nil, err
	}
	return &cert, nil
}

// parseRxBytes extracts received bytes for a peer from IPC get output.
func parseRxBytes(ipcOutput string, publicKey string) int64 {
	lines := strings.Split(ipcOutput, "\n")
	inPeer := false
	for _, line := range lines {
		line = strings.TrimSpace(line)
		if strings.HasPrefix(line, "public_key=") {
			inPeer = strings.TrimPrefix(line, "public_key=") == publicKey
			continue
		}
		if inPeer && strings.HasPrefix(line, "rx_bytes:") {
			val, err := strconv.ParseInt(strings.TrimPrefix(line, "rx_bytes:"), 10, 64)
			if err == nil {
				return val
			}
		}
	}
	return 0
}
